# E-ink Todo List — ESP8266_RTOS_SDK v3.4 rewrite (see ../.migration/04-rtos-architecture.md)
cmake_minimum_required(VERSION 3.5)

# PROJECT_VER from git describe at configure time — parity with the PlatformIO
# GIT_VERSION build flag. Lands in esp_app_desc_t.version and is consumed for
# the X-Device-Firmware-Version header and the launch screen.
execute_process(
    COMMAND git describe --tags --always --dirty
    WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}
    OUTPUT_VARIABLE EINKTODO_GIT_VERSION
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
    RESULT_VARIABLE EINKTODO_GIT_RESULT)
if(EINKTODO_GIT_RESULT EQUAL 0 AND NOT EINKTODO_GIT_VERSION STREQUAL "")
    set(PROJECT_VER "${EINKTODO_GIT_VERSION}")
else()
    set(PROJECT_VER "unknown")
endif()

# v3.4's project.cmake accepts only a single SDKCONFIG_DEFAULTS file (the
# semicolon-list support arrived in later IDF). Merge list entries into one
# generated file so the documented build incantation works:
#   idf.py -DSDKCONFIG_DEFAULTS="sdkconfig.defaults;sdkconfig.defaults.<env>" build
# kconfgen applies assignments in order, so later fragments override earlier
# ones (overlay semantics).
if(SDKCONFIG_DEFAULTS MATCHES ";")
    set(_merged_defaults "${CMAKE_BINARY_DIR}/sdkconfig.defaults.merged")
    file(WRITE "${_merged_defaults}" "# generated by rtos/CMakeLists.txt — do not edit\n")
    foreach(_fragment ${SDKCONFIG_DEFAULTS})
        get_filename_component(_fragment_abs "${_fragment}" ABSOLUTE
                               BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
        if(NOT EXISTS "${_fragment_abs}")
            message(FATAL_ERROR "SDKCONFIG_DEFAULTS fragment '${_fragment_abs}' does not exist.")
        endif()
        file(READ "${_fragment_abs}" _fragment_content)
        file(APPEND "${_merged_defaults}" "\n# --- ${_fragment} ---\n${_fragment_content}")
    endforeach()
    set(SDKCONFIG_DEFAULTS "${_merged_defaults}")
endif()

# v3.4 writes sdkconfig into the project root by default; with one build dir
# per overlay (and concurrent agents each using their own -B dir) that shared
# file would leak one overlay's values into the next configure (an existing
# sdkconfig overrides SDKCONFIG_DEFAULTS). Keep each build dir's sdkconfig
# private instead. project.cmake requires the file to exist when SDKCONFIG is
# set, so seed an empty one — confgen then fills it from the defaults files.
if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
endif()
if(NOT EXISTS "${SDKCONFIG}")
    file(WRITE "${SDKCONFIG}" "")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(einktodo-rtos)
